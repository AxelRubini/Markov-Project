17/* Resize table when load factor exceeds threshold.
18 * Rehashes all items and frees old buckets.
19 */
20static void ht_resize(hash_table_t *table) {
21 int new_size = next_prime(table->size * 2);
22 linked_list_t **old_buckets = table->buckets;
23 int old_size = table->size;
24
25 table->buckets = dmalloc(sizeof(linked_list_t *) * new_size); // new array
26 table->size = new_size;
27 for (int i = 0; i < new_size; ++i) {
28 table->buckets[i] = NULL;
29 }
30
31 /* Re‑hash every element into the new bucket array. */
32 for (int i = 0; i < old_size; ++i) {
33 linked_list_t *bucket = old_buckets[i];
34 if (bucket == NULL)
35 continue;
36
37 ll_item_t *current = bucket->head;
38 while (current) {
39 ht_item *it = (ht_item *)current->data;
40 int index = table->hash_func(get_ht_item_key(it), table->size);
41 linked_list_t *dest = ensure_bucket(table, index);
42 add_to_list(dest, it);
43 current = current->next;
44 }
45 /* Free the wrapper list; items are now in the new table. */
46 free(bucket);
47 }

80/* Insert item into table.
81 * Updates existing keys and triggers resize if needed.
82 * Ownership of item is taken by the table.
83 */
84void ht_insert(hash_table_t *table, ht_item *item) {
85 if (!table || !item)
86 return;
87
88 int index = table->hash_func(get_ht_item_key(item), table->size);
89 linked_list_t *bucket = ensure_bucket(table, index);
90
91 ll_item_t *current = bucket->head;
92 while (current) {
93 ht_item *existing = (ht_item *)current->data;
94 if (table->key_cmp(get_ht_item_key(existing), get_ht_item_key(item)) == 0) {
95 /* Key already present – update value using item's value. */
96 existing->update_value(existing, get_ht_item_value(item));
97 item->free_item(item); /* Item is redundant now. */
98 return;
99 }
100 current = current->next;
101 }
102
103 /* Key not present – append new item. */
104 add_to_list(bucket, item);
105 table->count++;
106
107 if (load_factor(table) > LOAD_FACTOR_THRESHOLD) {
108 ht_resize(table);
109 }
110}

133/* Remove item by key.
134 * Frees item via callback and adjusts count.
135 * Returns 1 on success, 0 if not found.
136 */
137int ht_remove(hash_table_t *table, const void *key) {
138 if (!table || !key)
139 return 0;
140
141 int index = table->hash_func(key, table->size);
142 linked_list_t *bucket = table->buckets[index];
143 if (bucket == NULL)
144 return 0;
145
146 ll_item_t *current = bucket->head;
147 ll_item_t *previous = NULL;
148 while (current) {
149 ht_item *it = (ht_item *)current->data;
150 if (table->key_cmp(get_ht_item_key(it), key) == 0) {
151 /* Unlink current. */
152 if (previous == NULL) {
153 bucket->head = current->next;
154 if (bucket->head == NULL)
155 bucket->tail = NULL;
156 } else {
157 previous->next = current->next;
158 if (current->next == NULL)
159 bucket->tail = previous;
160 }
161 if (it->free_item) {
162 it->free_item(it); // release item resources
163 }
164 free(current);
165 table->count--;
166 return 1; /* Removed. */
167 }
168 previous = current;
169 current = current->next;
170 }
171 return 0; /* Not found. */
