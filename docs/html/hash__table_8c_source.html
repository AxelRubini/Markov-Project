22/* Resize table when load factor exceeds threshold.
23 * Rehashes all items and frees old buckets.
24 */
25static void ht_resize(hash_table_t *table)
26{
27 uint32_t new_size = next_prime(table->size * 2);
28 linked_list_t **old_buckets = table->buckets;
29 uint32_t old_size = table->size;
30
31 table->buckets =
32 dmalloc(sizeof(linked_list_t *) * new_size); // new array
33 table->size = new_size;
34 for (size_t i = 0; i < new_size; ++i) {
35 table->buckets[i] = NULL;
36 }
37
38 /* Re‑hash every element into the new bucket array. */
39 for (size_t i = 0; i < old_size; ++i) {
40 linked_list_t *bucket = old_buckets[i];
41 if (bucket == NULL) {
42 continue;
43 }
44
45 ll_item_t *current = bucket->head;
46 while (current) {
47 ht_item *it = (ht_item *)current->data;
48 uint32_t index = table->hash_func(get_ht_item_key(it),
49 table->size);
50 linked_list_t *dest = ensure_bucket(table, index);
51 add_to_list(dest, it);
52 current = current->next;
53 }
54 /* Free the wrapper list; items are now in the new table. */
55 free(bucket);
56 }
57 free(old_buckets); // release old bucket array
58}

93/* Insert item into table.
94 * Updates existing keys and triggers resize if needed.
95 * Ownership of item is taken by the table.
96 */
97void ht_insert(hash_table_t *table, ht_item *item)
98{
99 if (!table || !item) {
100 return;
101 }
102
103 uint32_t index = table->hash_func(get_ht_item_key(item), table->size);
104 linked_list_t *bucket = ensure_bucket(table, index);
105
106 ll_item_t *current = bucket->head;
107 while (current) {
108 ht_item *existing = (ht_item *)current->data;
109 if (table->key_cmp(get_ht_item_key(existing),
110 get_ht_item_key(item)) == 0) {
111 /* Key already present – update value using item's
112 * value. */
113 existing->update_value(existing,
114 get_ht_item_value(item));
115 item->free_item(item); /* Item is redundant now. */
116 return;
117 }
118 current = current->next;
119 }
120
121 /* Key not present – append new item. */
122 add_to_list(bucket, item);
123 table->count++;
124
125 if (load_factor(table) > LOAD_FACTOR_THRESHOLD) {
126 ht_resize(table);
127 }
128}

158int ht_remove(hash_table_t *table, const void *key)

172 while (current) {
173 ht_item *it = (ht_item *)current->data;
174 if (table->key_cmp(get_ht_item_key(it), key) == 0) {
175 /* Unlink current. */

177 bucket->head = current->next;
178 if (bucket->head == NULL) {
179 bucket->tail = NULL;

189 }
190 free(current);

194 previous = current;
195 current = current->next;
