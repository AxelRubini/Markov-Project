172 output_csv, n_words, end};
173
174 // Parse the command line arguments
175 int nerrors = arg_parse(argc, argv, argtable);
176 if (nerrors > 0) {

178 return -1;

180
181 // Set run mode
182 if (sp->count > 0) {
183 out->mode = MODE_SP;
184 } else if (mp->count > 0) {
185 out->mode = MODE_MP;

187
188 // Set task
189 if (task1->count > 0) {
190 out->task = TASK_1;
191 } else if (task2->count > 0) {
192 out->task = TASK_2;
193 } else if (task_both->count > 0) {
194 out->task = TASK_BOTH;
195 }
196
197 // Set input and output files
198 out->input_txt = input_txt->filename[0];
199 out->input_csv = input_csv->filename[0];
200 out->output_txt = output_txt->filename[0];
201 out->output_csv = output_csv->filename[0];
202
203 // Set number of words and word to process
204 out->n_words = n_words->count > 0 ? n_words->ival[0] : 0;
205
206 // Free the argument table
207 arg_freetable(argtable, sizeof(argtable) / sizeof(argtable[0]));
