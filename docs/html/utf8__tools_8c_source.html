9/*
10 * Expected length of character sequence referenced by the leading byte.
11 * return 1 for ASCII characters (ASCII is Subset of UTF-8)
12 * return 2 for 2-byte sequences, 3 for 3-byte sequences, and
13 * 4 for 4-byte sequences.
14 */
15static unsigned utf8_length(u8 lead)
16{
17 if (lead < 0x80) {
18 return 1; // 0xxxxxxx
19 }
20 if ((lead >> 5) == 0x6) {
21 return 2; // 110xxxxx
22 }
23 if ((lead >> 4) == 0xE) {
24 return 3; // 1110xxxx
25 }
26 if ((lead >> 3) == 0x1E) {
27 return 4; // 11110xxx
28 }
29 return 0; // invalid
30}

32int utf8_getchar(void)
33{
34 int c = getchar();
35 if (c == EOF) {
36 return EOF; // end of input
37 }
38 u8 lead = (u8)c;
39
40 unsigned len = utf8_length(lead);
41 if (len == 1) {
42 return lead; // ASCII character
43 }
44
45 if (len < 2 || len > 4) {
46 return UTF8_ERROR; // invalid leading byte
47 }
48
49 uint32_t codepoint =
50 lead & ((1u << (7 - len)) - 1); // Mask to get the leading bits
51 for (unsigned i = 1; i < len; ++i) {
52 int ci = getchar();
53 if (ci == EOF) {
54 return EOF;
55 }
56 u8 byte = (u8)ci;
57 if ((byte >> 6) != 0x2) {
58 return UTF8_ERROR;
59 }
60 codepoint = (codepoint << 6) |
61 (byte & 0x3F); // append the continuation byte
62 }
63
64 return (int)codepoint;
65}

99int utf8_getchar_from_buf(const u8 *buf, size_t *offset)
100{
101 if (!buf || !offset) {
102 return UTF8_ERROR;
103 }
104
105 u8 lead = buf[*offset];
106 if (lead == '\0') {
107 return EOF; // end of buffer
108 }
109
110 uint32_t len = utf8_length(lead); // expect 1..4
111 if (len < 1 || len > 4) {
112 return UTF8_ERROR;
113 }
114
115 // Fast path: ASCII
116 if (len == 1) {
117 (*offset) += 1;
118 return (int)lead;
119 }
120
121 uint32_t cp = lead & ((1u << (7 - len)) - 1);
122
123 // Validate continuation bytes and assemble code point
124 for (unsigned i = 1; i < len; ++i) {
125 u8 c = buf[*offset + i];
126 if (c == '\0') {
127 return EOF; // truncated at end
128 }
129 if ((c & 0xC0) != 0x80) {
130 return UTF8_ERROR; // not 10xxxxxx
131 }
132 cp = (cp << 6) | (uint32_t)(c & 0x3F);
133 }
134
135 (*offset) += len;
136 return (int)cp;
137}
