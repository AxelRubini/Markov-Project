6/*
7 * Expected length of character sequence referenced by the leading byte.
8 * return 1 for ASCII characters (ASCII is Subset of UTF-8)
9 * return 2 for 2-byte sequences, 3 for 3-byte sequences, and
10 * 4 for 4-byte sequences.
11 */
12static unsigned utf8_length(u8 lead) {
13 if (lead < 0x80)
14 return 1; // 0xxxxxxx
15 if ((lead >> 5) == 0x6)
16 return 2; // 110xxxxx
17 if ((lead >> 4) == 0xE)
18 return 3; // 1110xxxx
19 if ((lead >> 3) == 0x1E)
20 return 4; // 11110xxx
21 return 0; // invalid
22}

24int utf8_getchar(void) {
25 int c = getchar();
26 if (c == EOF) {
27 return EOF; // end of input
28 }
29 u8 lead = (u8)c;
30
31 unsigned len = utf8_length(lead);
32 if (len == 1) {
33 return lead; // ASCII character
34 }
35
36 if (len < 2 || len > 4) {
37 return UTF8_ERROR; // invalid leading byte
38 }
39
40 uint32_t codepoint =
41 lead & ((1u << (7 - len)) - 1); // Mask to get the leading bits
42 for (unsigned i = 1; i < len; ++i) {
43 int ci = getchar();
44 if (ci == EOF)
45 return EOF;
46 u8 byte = (u8)ci;
47 if ((byte >> 6) != 0x2)
48 return UTF8_ERROR;
49 codepoint =
50 (codepoint << 6) | (byte & 0x3F); // append the continuation byte
51 }
52
53 return (int)codepoint;
54}

85int utf8_getchar_from_buf(const u8 *buf, size_t *offset) {
86 if (!buf || !offset)
87 return UTF8_ERROR;
88
89 u8 lead = buf[*offset];
90 if (lead == '\0')
91 return EOF; // end of buffer
92
93 uint32_t len = utf8_length(lead); // expect 1..4
94 if (len < 1 || len > 4)
95 return UTF8_ERROR;
96
97 // Fast path: ASCII
98 if (len == 1) {
99 (*offset) += 1;
100 return (int)lead;
101 }
102
103 uint32_t cp = lead & ((1u << (7 - len)) - 1);
104
105 // Validate continuation bytes and assemble code point
106 for (unsigned i = 1; i < len; ++i) {
107 u8 c = buf[*offset + i];
108 if (c == '\0')
109 return EOF; // truncated at end
110 if ((c & 0xC0) != 0x80)
111 return UTF8_ERROR; // not 10xxxxxx
112 cp = (cp << 6) | (uint32_t)(c & 0x3F);
113 }
114
115 (*offset) += len;
116 return (int)cp;
117}
