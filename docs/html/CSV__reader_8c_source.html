15 * Read a CSV line (ended with \n or EOF) from the file descriptor fd.
16 * *buf will be allocated or reallocated using cap if necessary.
17 * returns: bytes readed, or -1 on error, or 0 if end of file reached.
18 * NOTE: The double dereference of buf is used to allow the function to modify

22ssize_t read_csv_line(int fd, char **buf, size_t *cap) {
23 if (!buf || !cap || fd < 0) {
24 fprintf(stderr, "Invalid arguments to read_csv_line\n");
25 return -1;
26 }
27 if (*buf == NULL || *cap == 0) {
28 *cap = 1024; // Initial capacity
29 *buf = (char *)dmalloc(*cap);
30 }
31 size_t len = 0;
32 while (1) {
33 char ch;
34 ssize_t read_res = read(fd, &ch, 1);
35 if (read_res == 0) {
36 if (len == 0) {
37 (*buf)[0] = '\0'; // If no data read, return empty string
38 return 0; // End of file reached with no data read
39 }
40 break; // End of file reached, return what we have
41 }
42 if (read_res < 0) {
43 if (errno == EINTR) {
44 continue; // Interrupted, try again
45 }
46 fprintf(stderr, "Error reading from file descriptor %d\n", fd);
47 free(*buf);
48 *buf = NULL;
49 return -1; // Error reading
50 }
51 if (len + 1 >= *cap) {
52 size_t new_cap = (*cap < 4096)
53 ? (*cap * 2)
54 : (*cap + 4096); // increase capacity if greater than
55 // 4096 than plus 4096 not double
56 char *new_buf = (char *)realloc(*buf, new_cap);
57 if (new_buf == NULL) {
58 fprintf(stderr, "Memory reallocation failed for buffer\n");
59 free(*buf);
60 *buf = NULL;
61 return -1; // Error reallocating
62 }
63 *buf = new_buf;
64 *cap = new_cap;
65 }
66 if (ch == ' ' || ch == '\t') {
67 continue; // Skip whitespace characters
68 }
69 (*buf)[len++] = ch;
70 if (ch == '\n') {
71 break; // End of line reached
72 }
73 }
74
75 (*buf)[len] = '\0'; // Null-terminate the string
76 return (ssize_t)len; // Return the number of bytes read
77}
